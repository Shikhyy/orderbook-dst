```python
"""exchange_core/handlers/bankruptcy.py
Bankruptcy procedure after a liquidation: insurance draw, then loss sharing.
Rules R-L1 to R-L5. Target for task f8_adl_b.
"""
from exchange_core.types import Report


def handle_bankruptcy(
    engine: "Engine", account_id: str, pre_position: int  # type: ignore # noqa: F821
) -> list[Report]:
    """Cover the deficit of a just-liquidated account and return the reports in order."""
    state = engine.state
    account = state.accounts[account_id]

    # R-L1: Calculate deficit
    deficit = max(0, -account.free_cash)
    if deficit == 0:
        return []

    reports = []

    # R-L2: Fee vault covers first
    cover = min(deficit, max(0, state.fee_vault))
    if cover > 0:
        state.fee_vault -= cover
        account.free_cash += cover
        reports.append(Report(kind="INSURANCE_DRAWN", account=account_id, qty=cover))

    remaining = deficit - cover
    if remaining == 0:
        return reports

    # R-L3: Find eligible accounts (not LIQ, not X, position != 0, opposite sign to pre_position)
    eligible = []
    for acc_id, acc in state.accounts.items():
        if acc_id == "LIQ" or acc_id == account_id:
            continue
        if acc.position == 0:
            continue
        # Opposite sign check
        if pre_position > 0 and acc.position < 0:
            eligible.append((acc_id, acc))
        elif pre_position < 0 and acc.position > 0:
            eligible.append((acc_id, acc))
        # If pre_position == 0, no account has opposite sign (liquidation with 0 position shouldn't happen)

    if not eligible:
        # R-L4: No eligible accounts, bad debt
        reports.append(Report(kind="BAD_DEBT", account=account_id, qty=remaining))
        return reports

    # Calculate weights and total
    weights = {}
    total_weight = 0
    for acc_id, acc in eligible:
        weight = abs(acc.position)
        weights[acc_id] = weight
        total_weight += weight

    # Base shares (floor division)
    base_shares = {}
    sum_base = 0
    for acc_id, weight in weights.items():
        base = remaining * weight // total_weight
        base_shares[acc_id] = base
        sum_base += base

    leftover = remaining - sum_base

    # Sort for leftover distribution: descending weight, then ascending account id
    sorted_for_leftover = sorted(eligible, key=lambda x: (-weights[x[0]], x[0]))

    # Distribute leftover (one unit each to first `leftover` accounts)
    final_shares = base_shares.copy()
    for i in range(leftover):
        acc_id = sorted_for_leftover[i][0]
        final_shares[acc_id] += 1

    # Apply shares: X receives remaining, each eligible pays its share
    account.free_cash += remaining

    # Report ADL_APPLIED in ascending account-id order for accounts with share > 0
    for acc_id in sorted(final_shares.keys()):
        share = final_shares[acc_id]
        if share > 0:
            state.accounts[acc_id].free_cash -= share
            reports.append(Report(kind="ADL_APPLIED", account=acc_id, qty=share))

    return reports
```
