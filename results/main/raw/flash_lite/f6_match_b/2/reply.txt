"""exchange_core/handlers/new_limit.py
Handler for NEW_LIMIT events: validation, margin, matching and fill settlement.
Rules R-B4, R-B5, R-M1 to R-M5. Target for task f6_match_b.
"""
from exchange_core.handlers.settle import settle_fill
from exchange_core.types import Event, Report, Order, Fill
from exchange_core.intmath import ceil_div
from exchange_core import book, ledger

def handle_new_limit(engine: "Engine", event: Event) -> list[Report]:  # type: ignore
    state = engine.state
    
    # 1. Validation
    if event.account not in state.accounts:
        return [Report.rejected("UNKNOWN_ACCOUNT")]
    if event.qty <= 0 or event.price <= 0 or (event.price % engine.cfg.tick) != 0:
        return [Report.rejected("INVALID_ORDER")]
    
    # 3. Post-Only validation
    if event.flag == "POST_ONLY":
        would_match = False
        if event.side == "BUY":
            best = book.best_ask(state)
            if best is not None and best <= event.price:
                would_match = True
        else:
            best = book.best_bid(state)
            if best is not None and best >= event.price:
                would_match = True
        if would_match:
            return [Report.rejected("POST_ONLY_WOULD_CROSS")]
    
    # 4. Margin
    lock = ceil_div(event.qty * event.price * state.margin_bps, 10000)
    if state.accounts[event.account].free_cash < lock:
        return [Report.rejected("INSUFFICIENT_MARGIN")]
    
    # 5. Execution
    order_id = state.next_order_id
    state.next_order_id += 1
    prio_seq = state.next_prio_seq
    state.next_prio_seq += 1
    
    order = Order(
        order_id=order_id, account=event.account, side=event.side,
        price=event.price, qty=event.qty, cum_qty=0,
        prio_seq=prio_seq, margin_bps=state.margin_bps, lock=lock, flag=event.flag
    )
    
    ledger.lock_cash(state, event.account, lock)
    reports = [Report.accepted(order_id)]
    
    # Matching (R-M1 to R-M4)
    while order.leaves > 0:
        if order.side == "BUY":
            opp_orders = book.get_asks(state)
            if not opp_orders or opp_orders[0].price > order.price:
                break
        else:
            opp_orders = book.get_bids(state)
            if not opp_orders or opp_orders[0].price < order.price:
                break
        
        resting = opp_orders[0]
        
        # STP
        if resting.account == order.account:
            book.remove_order(state, resting.order_id)
            ledger.release_lock(state, resting.account, resting.lock)
            reports.append(Report.cancelled(resting.order_id, "STP"))
            continue
            
        # Fill
        fill_qty = min(order.leaves, resting.leaves)
        fill_price = resting.price
        
        fill = Fill(
            buy_order_id=order.order_id if order.side == "BUY" else resting.order_id,
            sell_order_id=resting.order_id if order.side == "BUY" else order.order_id,
            price=fill_price, qty=fill_qty, taker_order_id=order.order_id,
            buyer=order.account if order.side == "BUY" else resting.account,
            seller=resting.account if order.side == "BUY" else order.account
        )
        reports.extend(settle_fill(engine, fill))
        
        for o in [order, resting]:
            o.cum_qty += fill_qty
            old_lock = o.lock
            new_lock = ceil_div(o.leaves * o.price * o.margin_bps, 10000)
            o.lock = new_lock
            ledger.release_lock(state, o.account, old_lock - new_lock)
            
        if resting.leaves == 0:
            book.remove_order(state, resting.order_id)
            
    # Post-match
    if order.leaves > 0:
        if event.flag == "IOC":
            ledger.release_lock(state, order.account, order.lock)
            reports.append(Report.cancelled(order.order_id, "IOC_REMAINDER"))
        else:
            book.add_order(state, order)
            
    return reports
