## 2. Applicable Rules (docs/04_RULEBOOK.md Section 8)
- **R-W1 [Declared].** A PRICE_SHOCK event sets the `mark_price` on the state. The engine then runs the waterfall for each account except `LIQ`, in ascending account-id order.
- **R-W2 [Declared].** For an account with `position != 0`:
  - `avail = free_cash + position * mark_price`
  - `req = ceil_div(abs(position) * mark_price * MAINT_BPS, 10000)`
  - The account is in breach if `avail < req`.
  - An account with `position == 0` is never in breach.
- **R-W3 [Declared].** Stage 1. If the account is in breach, the engine cancels all resting orders of the account in ascending `prio_seq` order. Each cancel releases the lock to `free_cash` and reports `CANCELLED(order_id, RISK)`.
- **R-W4 [Declared].** After stage 1, the engine computes the breach test again with the new `free_cash`. If the account is not in breach, the engine must not liquidate.
- **R-W5 [Declared].** Stage 2. If the account is still in breach, the engine liquidates the whole position at `mark_price` against the account `LIQ`:
  - If `position > 0`: the account sells `position` units to `LIQ`. The account receives `position * mark_price`. `LIQ` pays the same amount.
  - If `position < 0`: the account buys `abs(position)` units from `LIQ`. The account pays `abs(position) * mark_price`. `LIQ` receives the same amount.
  - The position of the account becomes 0. The position of `LIQ` changes by the opposite amount (`liq.position += orig_position`).
  - No fee applies. The book does not change.
  - The engine reports `LIQUIDATED(account, qty)` where `qty` is the signed position before liquidation.
- **R-W6 [Declared].** The report order for PRICE_SHOCK is: `MARK_SET`, then for each account in ascending order the stage 1 cancels, then the `LIQUIDATED` report if it occurs.

```python
    cfg = EngineConfig()
    # A1 has 500M cash. Short 5 units at 100M -> received 500M cash -> cash = 1000M. Pos = -5.
    accounts = {"A1": 1_000_000_000, "A2": 1_000_000_000, "LIQ": 10_000_000_000}
    eng = Engine(cfg, accounts, mark_price=100_000_000)
    # Build positions: A1 SELL 5 to A2 BUY 5
    eng.process(Event(kind="NEW_LIMIT", account="A2", side="BUY", price=100_000_000, qty=5))
    eng.process(Event(kind="NEW_LIMIT", account="A1", side="SELL", price=100_000_000, qty=5))
    # A1 places resting BUY order locking margin: 1 unit @ 90M (lock = 18M)
    eng.process(Event(kind="NEW_LIMIT", account="A1", side="BUY", price=90_000_000, qty=1))
    o_id = eng.state.next_order_id - 1

    # Price shocks up to 190M.
    # A1 pos = -5. req = ceil(5 * 190M * 500 / 10000) = 47.5M.
    # avail before cancel: free_cash + (-5 * 190M).
    # Setting price so avail < req before cancel, but avail >= req after cancel releases 18M lock.
    eng.state.accounts["A1"].free_cash = 935_000_000  # avail = 935M - 950M = -15M < 47.5M (breach!)
    # After releasing 18M: free_cash = 953M. avail = 953M - 950M = 3M... let us set exact numbers:
    # We want: free_cash_before + (-5 * 190M) < 47.5M
    # and: free_cash_before + 18M + (-5 * 190M) >= 47.5M
    # -> free_cash_before - 950M < 47.5M  => free_cash_before < 997.5M
    # -> free_cash_before + 18M - 950M >= 47.5M => free_cash_before >= 979.5M
    eng.state.accounts["A1"].free_cash = 980_000_000

    reps = eng.process(Event(kind="PRICE_SHOCK", mark_price=190_000_000))

    assert Report.cancelled(o_id, "RISK") in reps
```

```python
"""exchange_core/handlers/shock.py"""
from exchange_core.book import get_account_orders, remove_order
from exchange_core.intmath import ceil_div
from exchange_core.types import Event, Report


def handle_shock(engine: "Engine", event: Event) -> list[Report]:  # type: ignore # noqa: F821
    """Process PRICE_SHOCK event and run margin waterfall.

    Rules R-W1 to R-W6:
    1. Set mark_price. Emit MARK_SET.
    2. Iterate over accounts (except LIQ) in ascending order.
    3. Stage 1: if breach, cancel resting orders in ascending prio_seq order.
    4. Stage 2: if still in breach, liquidate position against LIQ.
    """
    state = engine.state
    cfg = engine.cfg
    reports: list[Report] = []

    # Rule R-W1: set mark_price
    state.mark_price = event.mark_price
    reports.append(Report.mark_set(event.mark_price))

    # Rule R-W1: iterate accounts except LIQ in ascending account-id order
    sorted_accounts = sorted([a for a in state.accounts if a != "LIQ"])

    for acc_id in sorted_accounts:
        acc = state.accounts[acc_id]

        # Rule R-W2: position == 0 is never in breach
        if acc.position == 0:
            continue

        # Breach calculation
        avail = acc.free_cash + acc.position * state.mark_price
        req = ceil_div(abs(acc.position) * state.mark_price * cfg.maint_bps, 10_000)

        if avail >= req:
            continue

        # Rule R-W3: Stage 1 cancel resting orders in ascending prio_seq order
        orders = get_account_orders(state, acc_id)
        for o in orders:
            remove_order(state, o.order_id)
            acc.locked_margin -= o.lock
            acc.free_cash += o.lock
            o.lock = 0
            reports.append(Report.cancelled(o.order_id, "RISK"))


        # Rule R-W5: Stage 2 liquidate whole position at mark_price against LIQ
        orig_position = acc.position
        liq_account = state.accounts["LIQ"]

        if orig_position > 0:
            # Account sells position to LIQ
            transfer_amount = orig_position * state.mark_price
            acc.free_cash += transfer_amount
            liq_account.free_cash -= transfer_amount
            acc.position = 0
            liq_account.position += orig_position
        else:
            # Account buys abs(position) from LIQ
            transfer_amount = abs(orig_position) * state.mark_price
            acc.free_cash -= transfer_amount
            liq_account.free_cash += transfer_amount
            acc.position = 0
            liq_account.position += orig_position  # orig_position is negative

        reports.append(Report.liquidated(acc_id, orig_position))

    return reports
```

```python
    next_order_id: int
    next_prio_seq: int
    margin_bps: int
    mark_price: int
    sessions: dict[str, SessionState]
    initial_total: int

    def snapshot(self) -> "EngineState":
        """Produce an independent deep snapshot of the engine state."""
        return EngineState(
            accounts={k: v.clone() for k, v in self.accounts.items()},
            orders={k: Order(**v.__dict__) for k, v in self.orders.items()},
            fee_vault=self.fee_vault,
            next_order_id=self.next_order_id,
            next_prio_seq=self.next_prio_seq,
            margin_bps=self.margin_bps,
            mark_price=self.mark_price,
            sessions={k: v.clone() for k, v in self.sessions.items()},
            initial_total=self.initial_total,
        )

    def restore(self, snap: "EngineState") -> None:
        """Restore all state in-place from an earlier snapshot (Rule R-A4, ENG-08)."""
        self.accounts = {k: v.clone() for k, v in snap.accounts.items()}
        self.orders = {k: Order(**v.__dict__) for k, v in snap.orders.items()}
        self.fee_vault = snap.fee_vault
        self.next_order_id = snap.next_order_id
        self.next_prio_seq = snap.next_prio_seq
        self.margin_bps = snap.margin_bps
        self.mark_price = snap.mark_price
        self.sessions = {k: v.clone() for k, v in snap.sessions.items()}
        self.initial_total = snap.initial_total
```

```python
"""exchange_core/intmath.py
Integer arithmetic helpers.

Rule R-U1: All amounts, prices, and quantities are integers.
No other module does rounding.
No float is used. No bare / operator is used.
"""


def round_half_even(n: int, d: int) -> int:
    """Divide n by d and round to the nearest integer.

    Rule R-U2: When the result is exactly halfway, round to the even integer.

    Algorithm:
      Let q, r = divmod(n, d).
      If 2*r > d:  return q + 1.
      If 2*r == d: return q + (q % 2).   # q is odd -> +1, q is even -> +0
      Otherwise:   return q.

    Preconditions: n >= 0, d > 0.
    """
    q, r = divmod(n, d)
    if 2 * r > d:
        return q + 1
    if 2 * r == d:
        return q + (q % 2)
    return q


def ceil_div(n: int, d: int) -> int:
    """Return the smallest integer that is not below n / d.

    Rule R-U3: Equals -((-n) // d).

    Preconditions: n >= 0, d > 0.
    """
    return -((-n) // d)
```

```python
"""exchange_core/book.py
Book primitives. Derived views of resting orders in EngineState.
No rule logic belongs here (docs/03_ARCHITECTURE.md).

from typing import Optional

from exchange_core.state import EngineState
from exchange_core.types import Order


def add_order(state: EngineState, order: Order) -> None:
    """Add a resting order to the state."""
    state.orders[order.order_id] = order


def remove_order(state: EngineState, order_id: int) -> Optional[Order]:
    """Remove and return a resting order by order_id."""
    return state.orders.pop(order_id, None)


def get_order(state: EngineState, order_id: int) -> Optional[Order]:
    """Retrieve a resting order by order_id."""
    return state.orders.get(order_id)


def get_bids(state: EngineState) -> list[Order]:
    """Return all resting BUY orders in priority order (Rule R-B2)."""
    bids = [o for o in state.orders.values() if o.side == "BUY" and o.leaves > 0]
    bids.sort(key=lambda o: (-o.price, o.prio_seq))
    return bids


def get_asks(state: EngineState) -> list[Order]:
    """Return all resting SELL orders in priority order (Rule R-B2)."""
    asks = [o for o in state.orders.values() if o.side == "SELL" and o.leaves > 0]
    asks.sort(key=lambda o: (o.price, o.prio_seq))
    return asks


def best_bid(state: EngineState) -> Optional[int]:
    """Return highest bid price, or None if no bids exist."""
    bids = get_bids(state)
    return bids[0].price if bids else None


def best_ask(state: EngineState) -> Optional[int]:
    """Return lowest ask price, or None if no asks exist."""
    asks = get_asks(state)
    return asks[0].price if asks else None


def get_account_orders(state: EngineState, account: str) -> list[Order]:
    """Return all resting orders for an account, sorted by prio_seq ascending (Rule R-W3)."""
    orders = [o for o in state.orders.values() if o.account == account and o.leaves > 0]
    orders.sort(key=lambda o: o.prio_seq)
    return orders
```

```python
"""exchange_core/ledger.py
Cash and lock primitives.
No rule logic belongs here (docs/03_ARCHITECTURE.md).
"""
from typing import Optional

from exchange_core.state import Account, EngineState


def get_account(state: EngineState, account_id: str) -> Optional[Account]:
    """Retrieve an account by ID."""
    return state.accounts.get(account_id)


def has_account(state: EngineState, account_id: str) -> bool:
    """Return True if the account exists."""
    return account_id in state.accounts


def lock_cash(state: EngineState, account_id: str, amount: int) -> None:
    """Move amount from free_cash to locked_margin."""
    acc = state.accounts[account_id]
    acc.free_cash -= amount
    acc.locked_margin += amount


def release_lock(state: EngineState, account_id: str, amount: int) -> None:
    """Move amount from locked_margin to free_cash."""
    acc = state.accounts[account_id]
    acc.locked_margin -= amount
    acc.free_cash += amount


def transfer_cash(state: EngineState, from_id: str, to_id: str, amount: int) -> None:
    """Transfer free_cash from one account to another."""
    state.accounts[from_id].free_cash -= amount
    state.accounts[to_id].free_cash += amount


def adjust_position(state: EngineState, account_id: str, delta: int) -> None:
    """Adjust position by delta (positive = increase, negative = decrease)."""
    state.accounts[account_id].position += delta


def pay_fee(state: EngineState, account_id: str, fee: int) -> None:
    """Deduct fee from account free_cash and deposit to fee_vault."""
    state.accounts[account_id].free_cash -= fee
    state.fee_vault += fee


def pay_rebate(state: EngineState, account_id: str, rebate: int) -> None:
    """Credit rebate from fee_vault to account free_cash."""
    state.fee_vault -= rebate
    state.accounts[account_id].free_cash += rebate


def add_volume(state: EngineState, account_id: str, notional: int) -> None:
    """Add notional to an account's trading volume."""
    state.accounts[account_id].volume += notional
```
