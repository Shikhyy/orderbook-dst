```python
"""exchange_core/handlers/bankruptcy.py
Bankruptcy procedure after a liquidation: insurance draw, then loss sharing.
Rules R-L1 to R-L5. Target for task f8_adl_b.
"""
from exchange_core.types import Report


def handle_bankruptcy(
    engine: "Engine", account_id: str, pre_position: int  # type: ignore # noqa: F821
) -> list[Report]:
    """Cover the deficit of a just-liquidated account and return the reports in order."""
    state = engine.state
    acc_x = state.accounts[account_id]
    
    # R-L1: Deficit calculation
    deficit = max(0, -acc_x.free_cash)
    if deficit == 0:
        return []
    
    reports = []
    
    # R-L2: Insurance Draw
    cover = min(deficit, max(0, state.fee_vault))
    if cover > 0:
        state.fee_vault -= cover
        acc_x.free_cash += cover
        reports.append(Report(kind="INSURANCE_DRAWN", account=account_id, qty=cover))
    
    remaining = deficit - cover
    if remaining == 0:
        return reports

    # R-L3: Loss Sharing
    # Find eligible accounts: not LIQ, not X, position != 0, opposite sign of pre_position
    eligible_accounts = []
    total_weight = 0
    for aid, acc in state.accounts.items():
        if aid == "LIQ" or aid == account_id:
            continue
        if acc.position != 0:
            # Opposite sign: pre_position * acc.position < 0
            # Since we use integer arithmetic and position is non-zero:
            if (pre_position > 0 and acc.position < 0) or (pre_position < 0 and acc.position > 0):
                weight = abs(acc.position)
                eligible_accounts.append({"id": aid, "weight": weight})
                total_weight += weight

    if not eligible_accounts:
        # R-L4: Bad Debt
        reports.append(Report(kind="BAD_DEBT", account=account_id, qty=remaining))
        return reports

    # Calculate shares
    shares = {}
    sum_base = 0
    for item in eligible_accounts:
        base = (remaining * item["weight"]) // total_weight
        shares[item["id"]] = base
        sum_base += base
    
    leftover = remaining - sum_base
    # Sort for leftover distribution: descending weight, then ascending account id
    # We use a list of dictionaries/tuples to maintain association
    # Sort keys: (-weight, id)
    eligible_accounts.sort(key=lambda x: (-x["weight"], x["id"]))
    
    for i in range(leftover):
        aid = eligible_accounts[i]["id"]
        shares[aid] += 1
        
    # Apply shares and prepare reports
    # R-L3: Each eligible account pays its share from free_cash.
    # R-L3: For each eligible account with share > 0, in ascending account-id order, report ADL_APPLIED.
    adl_reports_data = []
    for aid, share in shares.items():
        if share > 0:
            state.accounts[aid].free_cash -= share
            adl_reports_data.append((aid, share))
            
    acc_x.free_cash += remaining
    
    # Sort ADL reports by ascending account-id order
    adl_reports_data.sort()
    for aid, share in adl_reports_data:
        reports.append(Report(kind="ADL_APPLIED", account=aid, qty=share))
        
    return reports
```
