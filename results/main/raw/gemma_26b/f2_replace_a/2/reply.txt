"""exchange_core/rules/replace_rule.py
Pure function for replace decisions.
Rules R-P1, R-P2 (steps 2, 3), and R-P3. Target for task f2_replace_a.
"""
from exchange_core.config import EngineConfig
from exchange_core.types import Order, ReplaceDecision


def decide_replace(
    order: Order,
    new_price: int,
    new_qty: int,
    cfg: EngineConfig,
) -> ReplaceDecision:
    """Decide whether a replace request is valid and whether it keeps priority.

    Rules R-P1, R-P2 (steps 2 and 3), R-P3.
    """
    # R-P2 Step 2: Check price validity
    if new_price <= 0 or new_price % cfg.TICK != 0:
        return ReplaceDecision(ok=False, reason="INVALID_ORDER", keep_priority=False, new_leaves=0)

    # R-P2 Step 3: Check quantity validity
    # new_qty is the new total order quantity (OrderQty).
    # It must be greater than the amount already filled (cum_qty).
    if new_qty <= order.cum_qty:
        return ReplaceDecision(ok=False, reason="INVALID_REPLACE", keep_priority=False, new_leaves=0)

    # Calculate new leaves: new total qty - already filled qty
    new_leaves = new_qty - order.cum_qty

    # R-P3: Priority logic
    # - If new_price == old price and new_qty < old qty: keeps priority.
    # - If new_price == old price and new_qty == old qty: nothing changes (keeps priority).
    # - Otherwise (price change or quantity increase): loses priority.
    # Note: new_qty < old qty is actually a special case of "price stays same".
    # If new_price == order.price AND new_qty <= order.qty, keep priority.
    # But the prompt says: "If new_price equals the old price and new_qty is less than the old qty, 
    # the order keeps its prio_seq... If new_price equals the old price and new_qty equals the old qty, 
    # nothing changes."
    # Combining these: If price is same AND new_qty <= old_qty, keep priority.
    # Wait, R-P2 Step 3 says new_qty must be > cum_qty.
    # So if cum_qty < new_qty <= order.qty, and new_price == order.price, it's keep priority.
    
    keep_priority = False
    if new_price == order.price and new_qty <= order.qty:
        keep_priority = True

    return ReplaceDecision(
        ok=True,
        reason="",
        keep_priority=keep_priority,
        new_leaves=new_leaves,
    )
