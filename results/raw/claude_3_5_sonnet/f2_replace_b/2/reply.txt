"""Order replace handler (Task f2_replace_b).

Rules:
- R-P1: Order replace carries order_id, new_price, new_qty (total size).
- R-P2: Rejection checks in order.
- R-P3: Priority preservation.
- R-P4: State updates.
- R-P5: Raw crossing validation without account filtering.
"""
from typing import Any

from exchange_core.book import best_ask, best_bid, get_order
from exchange_core.intmath import ceil_div
from exchange_core.ledger import get_account, lock_cash, release_lock
from exchange_core.types import Event, Report

try:
    """reference/f2_replace_a/replace_rule.py
Reference solution for task f2_replace_a.
"""
from exchange_core.config import EngineConfig
from exchange_core.types import Order, ReplaceDecision


def decide_replace(
    order: Order,
    new_price: int,
    new_qty: int,
    cfg: EngineConfig,
) -> ReplaceDecision:
    """Decide whether a replace request is valid and whether it keeps priority.

    Rules R-P1, R-P2 (steps 2 and 3), R-P3:
    - Step 2: If new_price <= 0 or new_price % TICK != 0: REJECTED(INVALID_ORDER)
    - Step 3: If new_qty <= cum_qty: REJECTED(INVALID_REPLACE)
    - Priority:
      - If new_price == old_price and new_qty < old_qty: keeps prio_seq
      - If new_price == old_price and new_qty == old_qty: nothing changes (keeps prio_seq)
      - In all other cases (price change or qty increase): gets new prio_seq
    """
    # Rule R-P2 step 2: price validation
    if new_price <= 0 or new_price % cfg.tick != 0:
        return ReplaceDecision(ok=False, reason="INVALID_ORDER", keep_priority=False, new_leaves=0)

    # Rule R-P2 step 3: quantity vs cum_qty
    if new_qty <= order.cum_qty:
        return ReplaceDecision(
            ok=False, reason="INVALID_REPLACE", keep_priority=False, new_leaves=0
        )

    new_leaves = new_qty - order.cum_qty

    # Rule R-P3: priority determination
    if new_price == order.price and new_qty <= order.qty:
        keep_priority = True
    else:
        keep_priority = False

    return ReplaceDecision(
        ok=True,
        reason="",
        keep_priority=keep_priority,
        new_leaves=new_leaves,
    )

except ImportError:
    from reference_alt_v2.rules.replace_rule import decide_replace


def handle_replace(engine: Any, event: Event) -> list[Report]:
    """Handle a REPLACE event.

    Rules R-P1, R-P2, R-P3, R-P4, R-P5:
    Rejection order:
    1. If order not resting: REJECTED(UNKNOWN_ORDER)
    2. If new_price <= 0 or new_price % TICK != 0: REJECTED(INVALID_ORDER)
    3. If new_qty <= cum_qty: REJECTED(INVALID_REPLACE)
    4. If new price would match opposite side of book (raw check, R-P5):
       REJECTED(REPLACE_WOULD_CROSS)
    5. If new_lock - old_lock > free_cash: REJECTED(INSUFFICIENT_MARGIN)

    If valid:
    - Update prio_seq if not keep_priority.
    - Update qty and price.
    - Adjust lock and account free_cash / locked_margin.
    - Report REPLACED(order_id).
    """
    order = get_order(engine.state, event.order_id)
    if order is None:
        return [Report.rejected("UNKNOWN_ORDER")]

    # Checks 2 & 3 via decide_replace
    decision = decide_replace(order, event.new_price, event.new_qty, engine.cfg)
    if not decision.ok:
        return [Report.rejected(decision.reason)]

    # Check 4: Crossing validation (R-P2 check 4, R-P5)
    # Checks raw best price on opposite side without filtering orders belonging to the same account.
    if order.side == "BUY":
        ask_price = best_ask(engine.state)
        if ask_price is not None and event.new_price >= ask_price:
            return [Report.rejected("REPLACE_WOULD_CROSS")]
    else:  # SELL
        bid_price = best_bid(engine.state)
        if bid_price is not None and event.new_price <= bid_price:
            return [Report.rejected("REPLACE_WOULD_CROSS")]

    # Check 5: Margin check (R-P2 check 5, R-P4)
    # Rate used is order.margin_bps stored in the order, NOT engine.state.margin_bps.
    new_leaves = decision.new_leaves
    new_lock = ceil_div(new_leaves * event.new_price * order.margin_bps, 10000)
    acc = get_account(engine.state, order.account)
    if acc is None:
        return [Report.rejected("UNKNOWN_ACCOUNT")]

    if new_lock > order.lock:
        lock_diff = new_lock - order.lock
        if acc.free_cash < lock_diff:
            return [Report.rejected("INSUFFICIENT_MARGIN")]

    # Valid replace: apply state changes (R-P3, R-P4)
    if not decision.keep_priority:
        order.prio_seq = engine.state.next_prio_seq
        engine.state.next_prio_seq += 1

    order.qty = event.new_qty
    order.price = event.new_price

    if new_lock > order.lock:
        lock_cash(engine.state, order.account, new_lock - order.lock)
    elif new_lock < order.lock:
        release_lock(engine.state, order.account, order.lock - new_lock)
    order.lock = new_lock

    return [Report.replaced(order.order_id)]
