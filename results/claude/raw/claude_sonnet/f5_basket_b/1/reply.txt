"""exchange_core/handlers/basket.py
Handler for atomic multi-leg basket orders.
Rules R-A1 to R-A5. Target for task f5_basket_b.
"""
from typing import Any

from exchange_core.types import Event, Report


def _leg_succeeded(flag: str, reports: list[Report]) -> bool:
    """Apply rule R-A3 to the reports produced by one leg."""
    accepted = False
    for rep in reports:
        if rep.kind == "REJECTED":
            return False
        if rep.kind == "ACCEPTED":
            accepted = True
    if not accepted:
        return False
    if flag == "IOC":
        for rep in reports:
            if rep.kind == "CANCELLED" and rep.reason == "IOC_REMAINDER":
                return False
    elif flag == "POST_ONLY":
        for rep in reports:
            if rep.kind == "FILL":
                return False
    return True


def handle_basket(engine: Any, event: Event) -> list[Report]:  # type: ignore # noqa: F821
    """Process a BASKET event with all-or-nothing rollback semantics."""
    state = engine.state
    snap = state.snapshot()
    collected: list[Report] = []

    for idx, leg in enumerate(event.legs):
        leg_event = Event(
            kind="NEW_LIMIT",
            account=event.account,
            side=leg.side,
            price=leg.price,
            qty=leg.qty,
            flag=leg.flag,
        )
        reports = engine.dispatch(leg_event)
        if not _leg_succeeded(leg.flag, reports):
            state.restore(snap)
            return [Report.basket_rolled_back(idx)]
        collected.extend(reports)

    collected.append(Report.basket_committed())
    return collected
